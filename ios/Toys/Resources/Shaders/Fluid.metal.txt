// Stable-fluids simulation (Jos Stam) on textures: advection, vorticity
// confinement, pressure projection. Velocity lives on a small grid in
// texels per second; pigment (dye) on a larger grid, as Kubelka-Munk
// absorption/scattering (K/S) weighted by amount, so blue and yellow make
// green as real paint does; a third grid holds the bristle grooves the
// brush leaves in the paint, carried by the flow like the pigment.
#include <metal_stdlib>
using namespace metal;

constexpr sampler linearSampler(coord::normalized, address::clamp_to_edge, filter::linear);

struct AdvectParams {
    float dt;
    float dissipation;
    float2 velTexel;
};

struct VorticityParams {
    float curl;
    float dt;
};

static inline bool outside(uint2 gid, texture2d<float, access::write> t) {
    return gid.x >= t.get_width() || gid.y >= t.get_height();
}

kernel void advect(texture2d<float, access::sample> velocity [[texture(0)]],
                   texture2d<float, access::sample> source [[texture(1)]],
                   texture2d<float, access::write> dest [[texture(2)]],
                   constant AdvectParams &p [[buffer(0)]],
                   uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    float2 size = float2(dest.get_width(), dest.get_height());
    float2 uv = (float2(gid) + 0.5) / size;
    float2 vel = velocity.sample(linearSampler, uv).xy;
    float2 coord = uv - p.dt * vel * p.velTexel;
    dest.write(source.sample(linearSampler, coord) * p.dissipation, gid);
}

// Every finger's splat of a frame in one pass (not one pass per splat).
struct Splat {
    float2 point;     // uv
    float2 dir;       // the groove's normal (grooves only)
    float4 value;     // velocity (xy), or pigment (K/S rgb, amount)
    float radius;
    float3 pad;
};

static inline float2 splatOffset(float2 uv, float2 point, float aspect) {
    float2 d = uv - point;
    d.x *= aspect;
    return d;
}

kernel void velocitySplats(texture2d<float, access::read> source [[texture(0)]],
                           texture2d<float, access::write> dest [[texture(1)]],
                           constant Splat *splats [[buffer(0)]],
                           constant uint &count [[buffer(1)]],
                           constant float &aspect [[buffer(2)]],
                           uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    float2 uv = (float2(gid) + 0.5) / float2(dest.get_width(), dest.get_height());
    float4 v = source.read(gid);
    for (uint i = 0; i < count; i++) {
        float2 d = splatOffset(uv, splats[i].point, aspect);
        float r2 = dot(d, d);
        if (r2 < splats[i].radius * 9.0) v += splats[i].value * exp(-r2 / splats[i].radius);
    }
    dest.write(v, gid);
}

// Paint is opaque: a new stroke covers what is under it (`cover`) instead of
// adding to it, so colours stay clean and only blend at the edges. The brush is
// a round tip with a defined edge (not a Gaussian), so a stroke keeps its width
// however long the finger stays.
kernel void paintSplats(texture2d<float, access::read> source [[texture(0)]],
                        texture2d<float, access::write> dest [[texture(1)]],
                        constant Splat *splats [[buffer(0)]],
                        constant uint &count [[buffer(1)]],
                        constant float &aspect [[buffer(2)]],
                        constant float &cover [[buffer(3)]],
                        uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    float2 uv = (float2(gid) + 0.5) / float2(dest.get_width(), dest.get_height());
    float4 v = source.read(gid);
    for (uint i = 0; i < count; i++) {
        float tip = sqrt(splats[i].radius) * 0.9;
        float dist = length(splatOffset(uv, splats[i].point, aspect));
        if (dist >= tip) continue;
        float s = 1.0 - smoothstep(tip * 0.55, tip, dist);
        v = v * (1.0 - cover * s) + splats[i].value * s;
    }
    // a ceiling on thickness: old paint can always be painted over
    dest.write(v / max(1.0, v.a / 3.0), gid);
}

// The brush's hairs leave parallel grooves along the stroke: stripes across
// the motion (`dir` is the unit normal to the stroke, in aspect-corrected uv)
// at a few spacings so they look irregular, blended over older grooves.
kernel void grooveSplats(texture2d<float, access::read> source [[texture(0)]],
                         texture2d<float, access::write> dest [[texture(1)]],
                         constant Splat *splats [[buffer(0)]],
                         constant uint &count [[buffer(1)]],
                         constant float &aspect [[buffer(2)]],
                         constant float &freq [[buffer(3)]],
                         uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    float2 uv = (float2(gid) + 0.5) / float2(dest.get_width(), dest.get_height());
    float v = source.read(gid).x;
    float2 a = float2(uv.x * aspect, uv.y);
    for (uint i = 0; i < count; i++) {
        float tip = sqrt(splats[i].radius) * 0.9;
        float dist = length(splatOffset(uv, splats[i].point, aspect));
        if (dist >= tip) continue;
        float g = 1.0 - smoothstep(tip * 0.5, tip, dist);
        float2 n = splats[i].dir;
        float across = dot(a, n);
        float along = dot(a, float2(-n.y, n.x));
        float stripes = sin(across * freq) * 0.5 + sin(across * freq * 1.63 + 1.3) * 0.3 + sin(across * freq * 0.55 + 2.1) * 0.35;
        stripes *= 0.75 + 0.25 * sin(along * freq * 0.13 + across * freq * 0.31);
        v = mix(v, stripes, clamp(g * 0.85, 0.0, 1.0));
    }
    dest.write(float4(v, 0, 0, 0), gid);
}

kernel void curl(texture2d<float, access::read> velocity [[texture(0)]],
                 texture2d<float, access::write> dest [[texture(1)]],
                 uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    int w = velocity.get_width(), h = velocity.get_height();
    int x = gid.x, y = gid.y;
    float L = velocity.read(uint2(max(x - 1, 0), y)).y;
    float R = velocity.read(uint2(min(x + 1, w - 1), y)).y;
    float T = velocity.read(uint2(x, min(y + 1, h - 1))).x;
    float B = velocity.read(uint2(x, max(y - 1, 0))).x;
    dest.write(float4(0.5 * (R - L - T + B), 0, 0, 0), gid);
}

kernel void vorticity(texture2d<float, access::read> velocity [[texture(0)]],
                      texture2d<float, access::read> curlTex [[texture(1)]],
                      texture2d<float, access::write> dest [[texture(2)]],
                      constant VorticityParams &p [[buffer(0)]],
                      uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    int w = curlTex.get_width(), h = curlTex.get_height();
    int x = gid.x, y = gid.y;
    float L = curlTex.read(uint2(max(x - 1, 0), y)).x;
    float R = curlTex.read(uint2(min(x + 1, w - 1), y)).x;
    float T = curlTex.read(uint2(x, min(y + 1, h - 1))).x;
    float B = curlTex.read(uint2(x, max(y - 1, 0))).x;
    float C = curlTex.read(gid).x;
    float2 force = 0.5 * float2(abs(T) - abs(B), abs(R) - abs(L));
    force /= length(force) + 0.0001;
    force *= p.curl * C;
    force.y *= -1.0;
    float2 vel = velocity.read(gid).xy + force * p.dt;
    vel = clamp(vel, -1000.0, 1000.0);
    dest.write(float4(vel, 0, 0), gid);
}

kernel void divergence(texture2d<float, access::read> velocity [[texture(0)]],
                       texture2d<float, access::write> dest [[texture(1)]],
                       uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    int w = velocity.get_width(), h = velocity.get_height();
    int x = gid.x, y = gid.y;
    float2 C = velocity.read(gid).xy;
    float L = x > 0 ? velocity.read(uint2(x - 1, y)).x : -C.x;
    float R = x < w - 1 ? velocity.read(uint2(x + 1, y)).x : -C.x;
    float T = y < h - 1 ? velocity.read(uint2(x, y + 1)).y : -C.y;
    float B = y > 0 ? velocity.read(uint2(x, y - 1)).y : -C.y;
    dest.write(float4(0.5 * (R - L + T - B), 0, 0, 0), gid);
}

kernel void scale(texture2d<float, access::read> source [[texture(0)]],
                  texture2d<float, access::write> dest [[texture(1)]],
                  constant float &factor [[buffer(0)]],
                  uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    dest.write(source.read(gid) * factor, gid);
}

kernel void pressure(texture2d<float, access::read> pressureTex [[texture(0)]],
                     texture2d<float, access::read> divergenceTex [[texture(1)]],
                     texture2d<float, access::write> dest [[texture(2)]],
                     uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    int w = pressureTex.get_width(), h = pressureTex.get_height();
    int x = gid.x, y = gid.y;
    float L = pressureTex.read(uint2(max(x - 1, 0), y)).x;
    float R = pressureTex.read(uint2(min(x + 1, w - 1), y)).x;
    float T = pressureTex.read(uint2(x, min(y + 1, h - 1))).x;
    float B = pressureTex.read(uint2(x, max(y - 1, 0))).x;
    float div = divergenceTex.read(gid).x;
    dest.write(float4((L + R + B + T - div) * 0.25, 0, 0, 0), gid);
}

kernel void gradientSubtract(texture2d<float, access::read> pressureTex [[texture(0)]],
                             texture2d<float, access::read> velocity [[texture(1)]],
                             texture2d<float, access::write> dest [[texture(2)]],
                             uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    int w = pressureTex.get_width(), h = pressureTex.get_height();
    int x = gid.x, y = gid.y;
    float L = pressureTex.read(uint2(max(x - 1, 0), y)).x;
    float R = pressureTex.read(uint2(min(x + 1, w - 1), y)).x;
    float T = pressureTex.read(uint2(x, min(y + 1, h - 1))).x;
    float B = pressureTex.read(uint2(x, max(y - 1, 0))).x;
    float2 vel = velocity.read(gid).xy - 0.5 * float2(R - L, T - B);
    dest.write(float4(vel, 0, 0), gid);
}

struct PaintVertex {
    float4 position [[position]];
    float2 uv;
};

vertex PaintVertex paintVertex(uint vid [[vertex_id]]) {
    float2 pos[3] = { float2(-1, -1), float2(3, -1), float2(-1, 3) };
    PaintVertex out;
    out.position = float4(pos[vid], 0, 1);
    out.uv = float2((pos[vid].x + 1) * 0.5, 1 - (pos[vid].y + 1) * 0.5);
    return out;
}

// Value noise for the paper's grain and the paint's texture.
static inline float hash21(float2 p) {
    p = fract(p * float2(123.34, 456.21));
    p += dot(p, p + 45.32);
    return fract(p.x * p.y);
}

static inline float valueNoise(float2 p) {
    float2 i = floor(p), f = fract(p);
    float2 u = f * f * (3.0 - 2.0 * f);
    float a = hash21(i), b = hash21(i + float2(1, 0)), c = hash21(i + float2(0, 1)), d = hash21(i + float2(1, 1));
    return mix(mix(a, b, u.x), mix(c, d, u.x), u.y);
}

// Baked once, at the drawable's size (px = pixels at 2 per point):
// r = cold-pressed paper height (soft bumps half a millimetre wide and a fine
// grain), g and b = slow noises that vary the paint's edge, a = the sheen's
// patchy roughness.
kernel void paperBake(texture2d<float, access::write> dest [[texture(0)]],
                      uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    float2 px = float2(gid) + 0.5;
    float grain = valueNoise(px * 0.42) * 0.6 + valueNoise(px * 1.1) * 0.4;
    float paper = valueNoise(px * 0.11) * 0.5 + valueNoise(px * 0.23 + 7.1) * 0.3 + grain * 0.2;
    float roughness = valueNoise(px * 0.06) * (smoothstep(0.6, 1.0, valueNoise(px * 0.3 + 9.0)) * 0.6 + 1.0) / 1.6;
    dest.write(float4(paper, valueNoise(px * 0.025), valueNoise(px * 0.04 + 3.7), roughness), gid);
}

/// 0...1 how much paint there is, from the pigment amount.
static inline float paintAmount(float a) { return 1.0 - exp(-a * 3.0); }

/// Where the paint layer is: a crisp edge whose bevel changes along the stroke
/// (steep here, a slow taper there), as real paint's edge does.
static inline float paintCoverage(float amount, float n, float m) {
    float lo = 0.26 + 0.12 * n;
    return smoothstep(lo, lo + 0.06 + 0.22 * m, amount);
}

// Each frame, on the pigment grid: the paint's surface. A layer (flat on top,
// not a tube), a little thicker where more paint piled up, the ridge a brush
// pushes up along the edge of its stroke, the bristles' grooves.
// r = height, g = layer (0 on bare paper).
kernel void surface(texture2d<float, access::read> dye [[texture(0)]],
                    texture2d<float, access::read> grooves [[texture(1)]],
                    texture2d<float, access::sample> paper [[texture(2)]],
                    texture2d<float, access::write> dest [[texture(3)]],
                    uint2 gid [[thread_position_in_grid]]) {
    if (outside(gid, dest)) return;
    float2 uv = (float2(gid) + 0.5) / float2(dest.get_width(), dest.get_height());
    float4 noise = paper.sample(linearSampler, uv);
    float a = dye.read(gid).a;
    float c = paintAmount(a);
    float layer = paintCoverage(c, noise.g, noise.b) * smoothstep(0.3, 0.7, c);
    float glob = 1.0 - exp(-a * 0.45);
    float ridge = smoothstep(0.3, 0.5, c) * (1.0 - smoothstep(0.55, 0.95, c));
    float groove = grooves.read(gid).x;
    float h = 0.5 * layer + 0.35 * glob + 0.035 * ridge + 0.11 * groove * layer;
    dest.write(float4(h, layer, 0, 0), gid);
}

/// Cubic B-spline sampling in four bilinear taps: smooth paint edges from the
/// coarse pigment grid, without the bilinear staircase.
static inline float4 sampleSmooth(texture2d<float, access::sample> t, float2 uv) {
    float2 size = float2(t.get_width(), t.get_height());
    float2 p = uv * size - 0.5;
    float2 i = floor(p), f = fract(p);
    float2 f2 = f * f, f3 = f2 * f;
    float2 w0 = (-f3 + 3.0 * f2 - 3.0 * f + 1.0) / 6.0;
    float2 w1 = (3.0 * f3 - 6.0 * f2 + 4.0) / 6.0;
    float2 w2 = (-3.0 * f3 + 3.0 * f2 + 3.0 * f + 1.0) / 6.0;
    float2 w3 = f3 / 6.0;
    float2 g0 = w0 + w1, g1 = w2 + w3;
    float2 h0 = (i - 1.0 + w1 / g0 + 0.5) / size;
    float2 h1 = (i + 1.0 + w3 / g1 + 0.5) / size;
    return g0.y * (g0.x * t.sample(linearSampler, float2(h0.x, h0.y)) + g1.x * t.sample(linearSampler, float2(h1.x, h0.y)))
         + g1.y * (g0.x * t.sample(linearSampler, float2(h0.x, h1.y)) + g1.x * t.sample(linearSampler, float2(h1.x, h1.y)));
}

static inline float3 fromKS(float3 ks) {
    return 1.0 + ks - sqrt(ks * ks + 2.0 * ks);
}

// The colour comes back from K/S (Kubelka-Munk, per channel), with a slight
// saturation lift; thin paint is a little translucent over the paper. Then the
// light: a soft key from the top left on the surface normal (the paint's from
// the surface pass, the paper's bumps from the baked paper), a GGX sheen with a
// patchy roughness, a coloured shade side, a shadow marched toward the light,
// darkening in the crease where paint meets paper, and a faint coloured bounce
// from the paint onto the paper next to it.
fragment float4 paintFragment(PaintVertex in [[stage_in]],
                              texture2d<float, access::sample> dye [[texture(0)]],
                              texture2d<float, access::sample> surfaceTex [[texture(1)]],
                              texture2d<float, access::read> paperTex [[texture(2)]]) {
    uint2 pp = uint2(min(in.position.xy, float2(paperTex.get_width() - 2, paperTex.get_height() - 2)));
    float4 paperHere = paperTex.read(pp);
    float paperRight = paperTex.read(pp + uint2(1, 0)).r, paperDown = paperTex.read(pp + uint2(0, 1)).r;

    // the edge from the smooth sample, the colour from the sharp one (mixes stay crisp)
    float4 d = sampleSmooth(dye, in.uv);
    float4 sharpDye = dye.sample(linearSampler, in.uv);
    float3 paperColor = float3(1.0, 0.985, 0.95);
    float3 color = fromKS(max(sharpDye.rgb / max(sharpDye.a, 1e-4), 0.0));
    float gray = dot(color, float3(0.299, 0.587, 0.114));
    color = clamp(gray + (color - gray) * 1.15, 0.0, 1.0);
    float amount = paintAmount(d.a);
    float coverage = paintCoverage(amount, paperHere.g, paperHere.b);
    float opacity = mix(0.82, 1.0, smoothstep(0.5, 0.85, amount));
    float3 base = mix(paperColor, color * mix(paperColor, float3(1.0), 0.6), coverage * opacity);

    float2 size = float2(dye.get_width(), dye.get_height());
    float2 texel = 1.5 / size;
    float2 sL = surfaceTex.sample(linearSampler, in.uv - float2(texel.x, 0)).rg;
    float2 sR = surfaceTex.sample(linearSampler, in.uv + float2(texel.x, 0)).rg;
    float2 sT = surfaceTex.sample(linearSampler, in.uv - float2(0, texel.y)).rg;
    float2 sB = surfaceTex.sample(linearSampler, in.uv + float2(0, texel.y)).rg;
    float layer = 0.25 * (sL.g + sR.g + sT.g + sB.g);
    float hHere = 0.25 * (sL.r + sR.r + sT.r + sB.r);
    // the paint's slope over three texels, the paper's over one pixel, softened under paint
    float paperRelief = mix(0.06, 0.012, layer) * 6.0 * 3.0;
    float2 slope = float2(sL.r - sR.r, sT.r - sB.r) * 6.0 + float2(paperHere.r - paperRight, paperHere.r - paperDown) * paperRelief;
    float3 n = normalize(float3(slope, 1.0));
    float3 light = normalize(float3(-0.45, -0.6, 0.75));
    float3 halfway = normalize(light + float3(0, 0, 1));
    float diffuse = clamp(dot(n, light) / light.z, 0.0, 1.6);
    float shade = mix(1.0, diffuse, mix(0.5, 0.7, coverage));

    // shadow: march toward the light with a rising ray; thicker paint above it shades this point
    float2 toLight = normalize(light.xy);
    float shadow = 0.0;
    for (int k = 1; k <= 4; k++) {
        float r = float(k) * 2.5;
        float h = surfaceTex.sample(linearSampler, in.uv + toLight * r / size).r;
        shadow = max(shadow, (h - (hHere + 0.035 * r)) * (1.15 - 0.15 * float(k)));
    }
    shadow = smoothstep(0.0, 0.3, shadow) * 0.24;
    // the crease (paint all around, lower here) and the paint's colour bouncing on the paper
    float around = 0.0;
    float3 bounce = float3(0.0);
    for (int k = 0; k < 4; k++) {
        float2 dir = float2(k == 0 ? 1 : (k == 1 ? -1 : 0), k == 2 ? 1 : (k == 3 ? -1 : 0));
        float4 nd = dye.sample(linearSampler, in.uv + dir * 6.0 / size);
        float nc = paintCoverage(paintAmount(nd.a), paperHere.g, paperHere.b);
        around += nc;
        bounce += fromKS(max(nd.rgb / max(nd.a, 1e-4), 0.0)) * nc;
    }
    float occlusion = clamp(around * 0.25 - coverage, 0.0, 1.0) * 0.09;
    float3 bounceColor = around > 0.01 ? bounce / around : paperColor;
    base = mix(base, base * mix(float3(1.0), bounceColor * 1.15, 0.12), (1.0 - coverage) * clamp(around * 0.25, 0.0, 1.0));

    // the sheen: GGX, its roughness patchy (wetter here, drying there), minus what a flat surface shows
    float roughness = mix(0.2, 0.42, paperHere.a);
    float alpha2 = roughness * roughness * roughness * roughness;
    float ndh = max(dot(n, halfway), 0.0), flatH = halfway.z;
    float dh = ndh * ndh * (alpha2 - 1.0) + 1.0, df = flatH * flatH * (alpha2 - 1.0) + 1.0;
    float sheen = max(0.0, alpha2 / (3.14159 * dh * dh) - alpha2 / (3.14159 * df * df)) * 0.035 * coverage;
    // light sinks a little into thick paint: its shade side stays coloured (deeper, not greyer)
    float s = clamp(shade, 0.68, 1.2);
    float3 body = s < 1.0 ? mix(pow(base, float3(1.0 + (1.0 - s) * 1.6)), base * s, 0.5) : base * s;
    float3 surfaceColor = mix(base * s, body, coverage);
    float3 lit = surfaceColor * (1.0 - shadow) * (1.0 - occlusion) + sheen;
    return float4(clamp(lit, 0.0, 1.0), 1.0);
}
